cmake_minimum_required(VERSION 3.24)
project(RosterCore VERSION 0.5.0 LANGUAGES CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/data/catalogues.json" ROSTERCORE_CATALOGUE_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/data/catalogues.json")
configure_file(src/catalogues.hpp.in generated/catalogues.hpp @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(nlohmann_json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nlohmann_json)
set(JSON_VALIDATOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json_schema URL https://github.com/pboettch/json-schema-validator/archive/refs/tags/2.4.0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json_schema)
find_package(ortools 9.15 CONFIG REQUIRED)
add_library(rostercore src/domain.cpp src/archives.cpp src/calendar_exports.cpp src/spacing.cpp src/change_sets.cpp src/calendar.cpp src/workforce.cpp src/evolution_validation.cpp src/verification.cpp src/validation.cpp src/solver.cpp src/history.cpp src/integrations.cpp src/hash.cpp)
target_include_directories(rostercore PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(rostercore PUBLIC nlohmann_json::nlohmann_json PRIVATE nlohmann_json_schema_validator ortools::ortools)
target_compile_definitions(rostercore PRIVATE ROSTERCORE_SCHEMA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/schemas")
if(MSVC)
 target_compile_options(rostercore PRIVATE /W4 /permissive- /utf-8)
else()
 target_compile_options(rostercore PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(cmake/RuntimeDlls.cmake)
add_executable(rostercore_cli src/cli.cpp)
set_target_properties(rostercore_cli PROPERTIES OUTPUT_NAME rostercore)
target_link_libraries(rostercore_cli PRIVATE rostercore)
rostercore_stage_runtime_dlls(rostercore_cli)
include(CTest)
if(BUILD_TESTING)
 add_executable(core_tests tests/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE rostercore)
 rostercore_stage_runtime_dlls(core_tests)
 add_test(NAME core_invariants COMMAND core_tests)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 add_test(NAME cli_startup COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_smoke.py $<TARGET_FILE:rostercore_cli>)
 set_tests_properties(cli_startup PROPERTIES FIXTURES_SETUP cli_ready TIMEOUT 30)
 add_test(NAME acceptance COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/acceptance.py $<TARGET_FILE:rostercore_cli>)
 add_test(NAME evolution COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/evolution.py $<TARGET_FILE:rostercore_cli>)
 add_test(NAME work_pattern_assignments COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/work_pattern_assignments.py $<TARGET_FILE:rostercore_cli>)
 add_test(NAME reliability COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reliability.py $<TARGET_FILE:rostercore_cli>)
 add_test(NAME archives COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/archives.py $<TARGET_FILE:rostercore_cli>)
 add_test(NAME spacing COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/spacing.py $<TARGET_FILE:rostercore_cli>)
 add_test(NAME change_sets COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/change_sets.py $<TARGET_FILE:rostercore_cli>)
 add_test(NAME calendar_exports COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/calendar_exports.py $<TARGET_FILE:rostercore_cli>)
 set_tests_properties(calendar_exports acceptance evolution work_pattern_assignments reliability archives spacing change_sets PROPERTIES FIXTURES_REQUIRED cli_ready)
endif()
if(APPLE)
 set_target_properties(rostercore_cli PROPERTIES INSTALL_RPATH "@loader_path/../lib")
elseif(UNIX)
 set_target_properties(rostercore_cli PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
install(TARGETS rostercore ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS rostercore_cli RUNTIME_DEPENDENCIES
 DIRECTORIES "${CMAKE_SOURCE_DIR}/.deps/ortools/lib" "${CMAKE_SOURCE_DIR}/.deps/ortools/bin"
 PRE_EXCLUDE_REGEXES "api-ms-.*" "ext-ms-.*"
 POST_EXCLUDE_REGEXES "^/lib/.*" "^/usr/lib/.*" "^/System/.*" ".*[Ww]indows[/\\].*"
 RUNTIME DESTINATION bin LIBRARY DESTINATION lib FRAMEWORK DESTINATION lib)
install(FILES LICENSE NOTICE docs/THIRD_PARTY_NOTICES.md docs/THIRD_PARTY_LICENSES.txt DESTINATION share/rostercore)
install(DIRECTORY "${CMAKE_SOURCE_DIR}/.deps/ortools/" DESTINATION share/rostercore/dependency-notices OPTIONAL
 FILES_MATCHING PATTERN "LICENSE*" PATTERN "NOTICE*" PATTERN "COPYING*"
 PATTERN "include" EXCLUDE PATTERN "lib" EXCLUDE PATTERN "bin" EXCLUDE PATTERN "examples" EXCLUDE)
install(DIRECTORY "${CMAKE_SOURCE_DIR}/.deps/json/" DESTINATION share/rostercore/dependency-notices/json OPTIONAL FILES_MATCHING PATTERN "LICENSE*" PATTERN "include" EXCLUDE PATTERN "tests" EXCLUDE)
install(FILES "${CMAKE_SOURCE_DIR}/.deps/json-schema-validator/LICENSE" DESTINATION share/rostercore/dependency-notices/json-schema-validator OPTIONAL)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY "${CMAKE_SOURCE_DIR}/.deps/json/include/nlohmann" DESTINATION include OPTIONAL)
install(DIRECTORY schemas/ DESTINATION share/rostercore/schemas)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_NAME rostercore)
include(CPack)
